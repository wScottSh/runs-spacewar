processor multiply_steps                                    # cla; mus x 021
  inout low: word
  in multiplicand: word
  out high: word
  set high = 0
  for step in 1 .. 021:
    set (high, low) = mus(high, low, multiplicand)

network multiply                                            # mpy
  writes: multiply_call
  owns high: word
  owns signs: word
  run magnitude
    value: multiply_call.entry
    magnitude: multiply_call.low
  run magnitude
    value: multiply_call.factor
    magnitude: multiply_call.partial
  run multiply_steps
    low: multiply_call.low
    multiplicand: multiply_call.partial
  run copy
    source: high
    target: multiply_call.partial
  run exclusive_or
    a: multiply_call.factor
    b: multiply_call.entry
    result: signs
  if signs < 0:
    run copy                                                # lac mp2: the xor took AC
      source: multiply_call.partial
      target: high
    run negate_double
      low: multiply_call.low
    run copy
      source: high
      target: multiply_call.partial
